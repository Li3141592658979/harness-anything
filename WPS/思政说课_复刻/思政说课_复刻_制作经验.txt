4. 核心架构范式
================================================================================
4.1 WPS COM 可见模式 + 数据驱动
    - win32com.client.Dispatch('KWPP.Application') → app.Visible = True
    - 生成前 taskkill /F /IM wps.exe /IM wpscenter.exe /IM wpp.exe /IM et.exe
      /IM etx.exe 清残留防 RPC 断连；connect_wps() 封装 4 次重试
    - 每页：s = pres.Slides.Add(idx, 12)（布局 12=空白）→
      s.Background.Fill.UserPicture(bg) 铺母版图 → 画内容
    - 每页后 app.ActiveWindow.View.GotoSlide(idx) + time.sleep(0.25)，让用户实时看到
    - 跑完不 app.Quit()，保留窗口供编辑
    - 导出：pres.SaveAs(pptx, 26) 与 pres.SaveAs(pdf, 32)；导出前先 os.remove 旧 PDF

4.2 标准 txt()（居中铁律）
    def txt(s, x, y, w, h, text, fs=13, color=INK, bold=False, align=1, font=FT, italic=False):
        tb = s.Shapes.AddTextbox(1, x, y, w, h)
        tf = tb.TextFrame
        tf.WordWrap = True          # 致命：False+居中会让文字跑出文本框
        try:
            tf.MarginLeft=0; tf.MarginRight=0; tf.MarginTop=0; tf.MarginBottom=0
            tf.AutoSize = 0
            tf.VerticalAnchor = 3   # msoAnchorMiddle 垂直居中
        except Exception: pass
        tf.TextRange.Text = text
        f = tf.TextRange.Font
        f.Name = font; f.Size = fs; f.Color.RGB = h2b(color)
        if bold: f.Bold = True
        if italic: f.Italic = True
        try: tf.TextRange.ParagraphFormat.Alignment = {1:1,2:2,3:3}[align]
        except Exception: pass
        return tb
    align: 1=左 / 2=中 / 3=右。居中、右对齐必须 WordWrap=True，否则文字左移越界。
    多行+垂直居中时 WPS 计算会上浮约 10pt：若标题顶端探出，改顶对齐或手工定 y。

4.3 左侧竖导航栏避让（本项目头等坑）
    原图左侧竖导航栏占 x[8,108] y[190,406]，列 7 个章节名。内容卡左缘若 <108
    会压住导航文字（用户原话「你的内容把侧边导航栏盖住了」）。
    修复：所有内容左缘统一 ≥118（导航右缘 108 + 10pt 安全距）。
    例：P2 教材图 x=118、卡片 x=322；P3 卡 x=122；P4 三列 x=118/398/678；
        P5 卡 x=122；P6 卡 x=122/398/674；P7 阶梯 x=122/392/662；P8 卡 x=122/498。
    内部元素随左缘同步右移、整体收窄，右缘 ≤940。导航用 pill(s,8,y,100,24,...)，
    激活态红底白字、非激活态灰字白底浅描边。
    校验：fitz 抽文字 bbox，若落在 x<108 & y[190,406] 且文本不在 7 章节名 → 侵入。

4.4 公共部件 chrome()
    每内容页开头 chrome(s, sec_name, active_tab)：
      1) 左上金星：shape(s,26,9,46,42,GOLD,st=92)，失败回退「★」文字
      2) 页标题深红胶囊：pill(s,84,10,196,40,sec_name,21,WHITE,REDD)
      3) 顶部 4 标签：激活态红底红字描边，非激活态白底黑字
      4) 左侧 7 章节竖导航（见 4.3）
    pill() = 圆角矩形(st=5)+文字；shape() = AddShape(st,...)，st：5=圆角矩形、9=椭圆、
    33=右箭头、92=五角星、18=圆环(donut)。

def h2b(c):
    return c[0] + c[1]*256 + c[2]*65536   # RGB -> BGR

TASKKILL = 'taskkill /F /IM wps.exe /IM wpscenter.exe /IM wpp.exe /IM et.exe /IM etx.exe >NUL 2>&1'

def connect_wps(retries=4):
    for i in range(retries):
        try:
            app = win32com.client.Dispatch('KWPP.Application')
            app.Visible = True
            _ = app.Presentations.Count
            return app
        except Exception:
            pythoncom.CoUninitialize()
            time.sleep(2.5)
    raise RuntimeError("WPS COM 连接失败")

app = connect_wps()
pres = app.Presentations.Add()
pres.PageSetup.SlideWidth = 960
pres.PageSetup.SlideHeight = 540

def new_slide(idx, bg):
    s = pres.Slides.Add(idx, 12)
    try:
        s.Background.Fill.UserPicture(bg)
    except Exception:
        s.FollowMasterBackground = False
        s.Background.Fill.UserPicture(bg)
    try:
        app.ActiveWindow.View.GotoSlide(idx)
    except Exception:
        pass
    time.sleep(0.25)
    return s

def txt(s, x, y, w, h, text, fs=13, color=INK, bold=False, align=1, font=FT, italic=False):
    tb = s.Shapes.AddTextbox(1, x, y, w, h)
    tf = tb.TextFrame
    tf.WordWrap = True
    try:
        tf.MarginLeft = 0; tf.MarginRight = 0; tf.MarginTop = 0; tf.MarginBottom = 0
        tf.AutoSize = 0
        tf.VerticalAnchor = 3
    except Exception:
        pass
    tf.TextRange.Text = text
    f = tf.TextRange.Font
    f.Name = font; f.Size = fs
    f.Color.RGB = h2b(color)
    if bold: f.Bold = True
    if italic: f.Italic = True
    try:
        tf.TextRange.ParagraphFormat.Alignment = {1: 1, 2: 2, 3: 3}[align]
    except Exception:
        pass
    return tb

def shape(s, x, y, w, h, fill, line=None, lw=1.0, st=1):
    sp = s.Shapes.AddShape(st, x, y, w, h)
    sp.Fill.Solid()
    sp.Fill.ForeColor.RGB = h2b(fill)
    if line is None:
        sp.Line.Visible = False
    else:
        sp.Line.ForeColor.RGB = h2b(line)
        sp.Line.Weight = lw
    return sp

def shape_text(sp, text, fs, color, bold=True, font=FT):
    tf = sp.TextFrame
    try:
        tf.MarginLeft = 0; tf.MarginRight = 0; tf.MarginTop = 0; tf.MarginBottom = 0
        tf.AutoSize = 0
        tf.VerticalAnchor = 3
    except Exception:
        pass
    tf.WordWrap = True
    tf.TextRange.Text = text
    f = tf.TextRange.Font
    f.Name = font; f.Size = fs
    f.Color.RGB = h2b(color)
    if bold: f.Bold = True
    try:
        tf.TextRange.ParagraphFormat.Alignment = 2
    except Exception:
        pass

def pill(s, x, y, w, h, text, fs, fg, bg, bold=True, line=None, lw=1.0, font=FT):
    sp = shape(s, x, y, w, h, bg, line=line, lw=lw, st=5)
    shape_text(sp, text, fs, fg, bold=bold, font=font)
    return sp

def pic(s, path, x, y, w, h):
    return s.Shapes.AddPicture(os.path.normpath(path), 0, 1, x, y, w, h)

7.1 原始模板背景（项目早期）
    早期用 ImageGen 生成红金党建风底图：cover_bg.png（华表+红绸+书法空区）、

7.2 换成用户新图并去水印（当前生效，路径自动定位）
    把用户给的新背景原图放到 work/new_bg.jpg（任意名称，改 SRC 即可），
    运行 make_new_bg.py 生成 bgs/bg_cover.png 与 bgs/bg_content.png。

    # -*- coding: utf-8 -*-
    """新背景图处理：去水印 -> 裁 16:9 -> 封面版 + 内容页柔白蒙版版（跨电脑通用）"""
    from PIL import Image, ImageDraw, ImageFilter
    import os

    HERE = os.path.dirname(os.path.abspath(__file__))            # .../work
    BGS  = os.path.join(HERE, "bgs")
    SRC  = os.path.join(HERE, "new_bg.jpg")   # 把用户新背景图放 work/ 下改名 new_bg.jpg
    # 若新图是别的名字/位置，直接改上面 SRC（支持绝对路径或相对本脚本的路径）

    im = Image.open(SRC).convert('RGB')
    w, h = im.size
    print("原图:", w, h)

    # 1) 去水印：右下角白字，用左侧相邻同色纹理 patch 覆盖（左缘羽化）
    #    —— 不同图的水印位置不同，按实际坐标改下面 4 个数
    wx0, wy0, wx1, wy1 = int(w*0.897), int(h*0.917), w, h   # 默认：右下角 10% 区域
    pw = wx1 - wx0
    patch = im.crop((wx0 - pw, wy0, wx0, wy1))
    mask = Image.new('L', patch.size, 255)
    d = ImageDraw.Draw(mask)
    for i in range(30):
        d.line([(i, 0), (i, patch.size[1])], fill=int(255 * i / 30))
    im.paste(patch, (wx0, wy0), mask)
    print("水印已去除:", (wx0, wy0, wx1, wy1))

    # 2) 裁剪 16:9：检测华表金色最高点，顶部少切保狮子，其余切底部
    px = im.load()
    gold_min_y = h
    for y in range(0, 300, 2):
        hit = False
        for x in range(860, 1250, 3):
            if x >= w: break
            r, g, b = px[x, y]
            if r > 170 and g > 115 and b < 150 and r - b > 55:
                hit = True
                break
        if hit:
            gold_min_y = y
            break
    top = max(0, gold_min_y - 14)
    if top + 864 > h:
        top = h - 864
    im2 = im.crop((0, top, w, top + 864)).resize((960, 540), Image.LANCZOS)
    im2.save(os.path.join(BGS, "bg_cover.png"))
    print("封面版完成: gold_min_y=%d top=%d" % (gold_min_y, top))

    # 3) 内容页：柔边白蒙版盖"标题+内容区"(85,46)-(952,460)，四边保留氛围
    white = Image.new('RGB', im2.size, (255, 255, 255))
    m = Image.new('L', im2.size, 0)
    d = ImageDraw.Draw(m)
    d.rectangle((85, 46, 952, 460), fill=215)
    m = m.filter(ImageFilter.GaussianBlur(36))
    im3 = Image.composite(white, im2, m)
    im3.save(os.path.join(BGS, "bg_content.png"))
    print("内容页版完成")

    原理：
    - 去水印：水印是规则白字，取左侧相邻同色纹理 patch 覆盖，左缘 30px 线性渐变
      mask 羽化接缝，肉眼无痕。不同图改 wx0,wy0,wx1,wy1。
    - 自动定裁剪线：扫描 y<300 金色像素最高点 → top=gold_min_y-14，裁 top:top+864
      即 16:9，LANCZOS 缩到 960×540，保住华表顶部狮子。
    - 内容页蒙版：只在标题+内容区叠柔白（fill=215 + GaussianBlur(36)），四周保留。
      blur≈36 最自然；fill 越大越白。
    旧背景备份 bg_cover_old.png / bg_content_old.png。build_sz.py 布局零改动。
    换全幅背景口诀：「特征像素最高点定裁剪线 + 内容区柔白蒙版 + 自带底色元素不保护」。

9. fitz 程序化校验（每次生成后必跑，路径自动定位）
================================================================================
存为 work/check.py，运行 python check.py：
    import os, fitz
    HERE = os.path.dirname(os.path.abspath(__file__))
    BASE = os.path.dirname(HERE)
    pdf = os.path.join(BASE, "思政说课_1比1复刻.pdf")
    doc = fitz.open(pdf)
    W,H,TOL = 960,540,1.0
    bad = []; nav_hits = []
    NAV_NAMES = ['课程介绍','学情分析','课前准备','能力提升','价值认同','教学反思','创新方向']
    for pi in range(len(doc)):
        for b in doc[pi].get_text('blocks'):
            x0,y0,x1,y1 = b[:4]
            if x0 < -TOL or y0 < -TOL or x1 > W+TOL or y1 > H+TOL:
                bad.append((pi+1, round(x0,1),round(y0,1),round(x1,1),round(y1,1), b[4][:16]))
            if x0 < 108-TOL and y0 > 190-TOL and y1 < 406+TOL:
                t = b[4].strip()
                if t not in NAV_NAMES:
                    nav_hits.append((pi+1, t[:16]))
    print('文字越界:', len(bad), bad[:5])
    print('导航区侵入(非导航文字):', len(nav_hits), nav_hits[:5])
    for pn in [1,2,5,7]:
        doc[pn-1].get_pixmap(dpi=96).save(os.path.join(HERE, f"chk_p{pn}.png"))
验收标准：文字越界==0 且 导航区侵入==0。然后目检 P1/P2/P5/P7。

11. 1:1 复刻方法论
================================================================================
1) 量比例：截图像素尺寸与 960×540 等比缩放，元素位置按「截图占比 × 画布」换算
2) 定保护区：先识别校名/校徽/导航/页脚固定矩形（本项目左侧导航 x[8,108] y[190,406]）

4) 先布骨架后填字：先画卡片/图片盒子，目测对齐后再放文字

6) 校验双保险：fitz 查越界/导航侵入（量化）+ 关键页目检（定性）
